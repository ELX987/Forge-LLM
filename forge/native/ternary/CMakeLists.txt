cmake_minimum_required(VERSION 3.18)
project(tern158 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
if(NOT CMAKE_CUDA_ARCHITECTURES)
  # e.g. -DCMAKE_CUDA_ARCHITECTURES="80;86;89;90;100;120"  (CMake >= 3.24 also accepts "native")
  set(CMAKE_CUDA_ARCHITECTURES "80;86;89;90")
endif()
file(GLOB T158_SRC kernel/src/*.cu)
add_library(tern158 SHARED ${T158_SRC})
target_include_directories(tern158 PUBLIC kernel/include)
target_compile_options(tern158 PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr -lineinfo>)
find_package(CUDAToolkit REQUIRED)
target_link_libraries(tern158 PUBLIC CUDA::cublas CUDA::cudart)
add_executable(tern158_selftest kernel/tests/selftest.cu)
target_link_libraries(tern158_selftest PRIVATE tern158)
